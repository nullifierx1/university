## A. 🏁 Roundtrip — Траектория Свободы

```
// ==========================================================================
//  DSU (Disjoint Set Union) — система непересекающихся множеств
//
//  Используется дважды:
//  1) dsuHalf: объединяет полурёбра в подциклы (на основе спаривания)
//  2) dsuComp: объединяет компоненты при построении MST (Kruskal)
//
//  Оптимизации: сжатие путей + объединение по рангу → O(α(n)) на запрос
// ==========================================================================
struct DSU {
    std::vector<int> p, sz;
    DSU() {}
    DSU(int n) : p(n), sz(n, 1) { std::iota(p.begin(), p.end(), 0); }

    // find — найти корень множества с сжатием путей
    int find(int x) {
        return p[x] == x ? x : p[x] = find(p[x]);
    }

    // unite — объединить множества; возвращает true если были разными
    bool unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;           // уже в одном множестве
        if (sz[a] < sz[b]) std::swap(a, b); // меньшее к большему
        p[b] = a;
        sz[a] += sz[b];
        return true;
    }
};

struct Point {
    long double x, y;
};

// ==========================================================================
//  V4 — информация о вершине степени 4
//
//  h[4]    — четыре полуребра, исходящих из этой вершины
//  cost[3] — стоимость (сумма углов) каждого из 3 вариантов спаривания
//  best    — индекс варианта с минимальной стоимостью
// ==========================================================================
struct V4 {
    std::array<int, 4> h;            // полурёбра из вершины
    int best;                        // индекс лучшего спаривания (0, 1, или 2)
    std::array<long double, 3> cost; // стоимости трёх вариантов
};

// ==========================================================================
//  Edge — ребро в графе компонент для MST
//
//  u, v — номера компонент (подциклов)
//  w    — стоимость переключения (Δ = альтернатива − лучшее)
// ==========================================================================
struct Edge {
    int u, v;
    long double w;
    bool operator<(const Edge& other) const {
        return w < other.w;
    }
};

// ==========================================================================
//  Три варианта спаривания четырёх полурёбер {0, 1, 2, 3}:
//
//  P[0]: пары (0,1) и (2,3)
//  P[1]: пары (0,2) и (1,3)
//  P[2]: пары (0,3) и (1,2)
//
//  Каждая строка P[t] = {a, b, c, d} задаёт пары (h[a],h[b]) и (h[c],h[d])
// ==========================================================================
const int P[3][4] = {
    {0, 1, 2, 3},
    {0, 2, 1, 3},
    {0, 3, 1, 2}
};

int main() {
    std::ios::sync_with_stdio(false);
    std::cin.tie(nullptr);

    const long double EPS = 1e-18L;  // для сравнения long double
    int n, m;
    std::cin >> n >> m;

    // ──────────────────────────────────────────────────────────────────────
    //  Чтение координат вершин
    // ──────────────────────────────────────────────────────────────────────
    std::vector<Point> pt(n);
    for (int i = 0; i < n; ++i) {
        std::cin >> pt[i].x >> pt[i].y;
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Построение полурёберного представления (half-edge structure)
    //
    //  Каждое ребро e (между u и v) порождает два полуребра:
    //    h₁ = 2e:     hv[h₁] = u,  to[h₁] = v   (из u в v)
    //    h₂ = 2e+1:   hv[h₂] = v,  to[h₂] = u   (из v в u)
    //    opp[h₁] = h₂,  opp[h₂] = h₁             (противоположные)
    //
    //  inc[v] — список полурёбер, ИСХОДЯЩИХ из вершины v
    //           (|inc[v]| = degree(v) = 2 или 4)
    // ──────────────────────────────────────────────────────────────────────
    std::vector<int> hv(2 * m), to(2 * m), opp(2 * m);
    std::vector<std::vector<int>> inc(n);

    for (int e = 0; e < m; ++e) {
        int u, v;
        std::cin >> u >> v;

        int h1 = 2 * e, h2 = 2 * e + 1;
        hv[h1] = u; to[h1] = v; opp[h1] = h2;    // полуребро из u в v
        hv[h2] = v; to[h2] = u; opp[h2] = h1;    // полуребро из v в u

        inc[u].push_back(h1);   // u → v
        inc[v].push_back(h2);   // v → u
    }

    // ──────────────────────────────────────────────────────────────────────
    //  turn(h1, h2) — угол поворота при проезде через вершину
    //
    //  h1 и h2 — два полуребра, исходящих из одной вершины v.
    //  Маршрут: ... → to[h1] → v → to[h2] → ...
    //  (приехали из to[h1], уезжаем в to[h2])
    //
    //  Направление прибытия: от to[h1] к v = -(to[h1] - v) = (v - to[h1])
    //  Направление отъезда:  от v к to[h2] = (to[h2] - v)
    //
    //  Угол поворота = угол между этими двумя направлениями:
    //    atan2(|cross|, -dot) где:
    //      cross = (a-v) × (b-v)   [векторное произведение]
    //      dot   = (a-v) · (b-v)   [скалярное произведение]
    //      a = to[h1],  b = to[h2]
    //
    //  Знак: −dot потому что прибытие идёт в направлении −(a−v),
    //         а отъезд в направлении (b−v).
    //
    //  Результат:  0  = прямо (a, v, b на линии, b за v)
    //              π  = разворот (b = a)
    // ──────────────────────────────────────────────────────────────────────
    auto turn = [&](int h1, int h2) {
        int v = hv[h1], a = to[h1], b = to[h2];
        long double ax = pt[a].x - pt[v].x;
        long double ay = pt[a].y - pt[v].y;
        long double bx = pt[b].x - pt[v].x;
        long double by = pt[b].y - pt[v].y;
        long double cross = fabsl(ax * by - ay * bx);  // |sin θ| · |a| · |b|
        long double dot = ax * bx + ay * by;           //  cos θ  · |a| · |b|
        return std::atan2(cross, -dot);                 // угол ∈ [0, π]
    };

    // ──────────────────────────────────────────────────────────────────────
    //  Обработка каждой вершины: выбор лучшего спаривания
    //
    //  partner[h] — полуребро, спаренное с h в текущем выборе
    //  (приехав по h, уезжаем по partner[h], и наоборот)
    //
    //  base — суммарный угол при выборе лучшего спаривания в каждой вершине
    //  v4   — массив вершин степени 4 (для дальнейшей обработки)
    // ──────────────────────────────────────────────────────────────────────
    std::vector<int> partner(2 * m, -1);
    std::vector<V4> v4;
    long double base = 0;

    for (int v = 0; v < n; ++v) {
        if (inc[v].size() == 2) {
            // ── Вершина степени 2: единственный вариант спаривания ──
            //
            // Два полуребра a и b спариваются друг с другом.
            // Поворот неизбежен — добавляем к base.
            int a = inc[v][0], b = inc[v][1];
            partner[a] = b;
            partner[b] = a;
            base += turn(a, b);
        } else {
            // ── Вершина степени 4: три варианта спаривания ──
            //
            // Перебираем все 3 совершенных паросочетания на 4 полурёбрах.
            // Для каждого считаем суммарный поворот (2 поворота на вершину).
            // Выбираем вариант с минимальной суммой.
            V4 cur;
            for (int i = 0; i < 4; ++i) cur.h[i] = inc[v][i];

            // Вычисляем стоимость каждого из 3 спариваний
            for (int t = 0; t < 3; ++t) {
                int a = cur.h[P[t][0]], b = cur.h[P[t][1]];  // первая пара
                int c = cur.h[P[t][2]], d = cur.h[P[t][3]];  // вторая пара
                cur.cost[t] = turn(a, b) + turn(c, d);       // сумма поворотов
            }

            // Находим вариант с минимальной стоимостью
            cur.best = 0;
            for (int t = 1; t < 3; ++t) {
                if (cur.cost[t] + EPS < cur.cost[cur.best]) cur.best = t;
            }

            // Применяем лучшее спаривание: устанавливаем партнёров
            int t = cur.best;
            int a = cur.h[P[t][0]], b = cur.h[P[t][1]];
            int c = cur.h[P[t][2]], d = cur.h[P[t][3]];

            partner[a] = b; partner[b] = a;   // первая пара
            partner[c] = d; partner[d] = c;   // вторая пара

            base += cur.cost[t];   // добавляем стоимость лучшего варианта
            v4.push_back(cur);     // сохраняем для дальнейшего анализа
        }
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Определение подциклов через DSU
    //
    //  Спаривание partner + opposite определяют цепочки полурёбер:
    //    h → opp[h] → partner[opp[h]] → opp[partner[opp[h]]] → ...
    //
    //  Полурёбра в одной цепочке = один подцикл эйлерова обхода.
    //
    //  Правила объединения:
    //    1) h ~ opp[h]:     одно физическое ребро, проходится один раз
    //    2) h ~ partner[h]: спарены в вершине, последовательны в цикле
    //
    //  После DSU: каждая компонента = один подцикл.
    //  Для корректного эйлерова цикла нужна ОДНА компонента (k = 1).
    // ──────────────────────────────────────────────────────────────────────
    DSU dsuHalf(2 * m);
    for (int h = 0; h < 2 * m; ++h) {
        dsuHalf.unite(h, opp[h]);       // одно ребро → один подцикл
        dsuHalf.unite(h, partner[h]);   // спаренные → один подцикл
    }

    // Нумеруем компоненты (подциклы) от 0 до k-1
    std::vector<int> id(2 * m, -1), comp(2 * m);
    int k = 0;   // количество подциклов
    for (int h = 0; h < 2 * m; ++h) {
        int r = dsuHalf.find(h);
        if (id[r] == -1) id[r] = k++;   // новая компонента
        comp[h] = id[r];                // номер подцикла для полуребра h
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Построение графа компонент для MST
    //
    //  Для каждой вершины степени 4:
    //    • Лучшее спаривание t соединяет пары (a,b) и (c,d)
    //    • comp[a] = comp[b] (они спарены → один подцикл)
    //    • comp[c] = comp[d] (аналогично)
    //    • Если comp[a] ≠ comp[c] — это два РАЗНЫХ подцикла
    //
    //  Переключение на альтернативное спаривание i ≠ t:
    //    • Перепаривает полурёбра, объединяя подциклы comp[a] и comp[c]
    //    • Дополнительная стоимость: Δ = cost[i] − cost[t] ≥ 0
    //
    //  Создаём ребро (comp[a], comp[c]) с весом min{Δ} в графе компонент.
    //  MST на этом графе = минимальная стоимость объединения всех подциклов.
    //
    //  ┌─────────────┐   Δ = min(cost[i] − cost[t])   ┌─────────────┐
    //  │  Подцикл X  │ ────────────────────────────── │  Подцикл Y  │
    //  │  (comp[a])  │                                 │  (comp[c])  │
    //  └─────────────┘                                 └─────────────┘
    // ──────────────────────────────────────────────────────────────────────
    std::vector<Edge> edges;
    for (const auto& cur : v4) {
        int t = cur.best;
        int a = cur.h[P[t][0]];   // представитель первой пары
        int c = cur.h[P[t][2]];   // представитель второй пары
        int x = comp[a], y = comp[c];

        // Если обе пары уже в одном подцикле — переключение бесполезно
        if (x == y) continue;

        // Минимальная доплата за объединение через эту вершину
        long double add = 1e100L;
        for (int i = 0; i < 3; ++i) {
            if (i != t) add = std::min(add, cur.cost[i] - cur.cost[t]);
        }
        // Защита от ошибок округления
        if (add < 0 && add > -1e-15L) add = 0;

        edges.push_back({x, y, add});
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Алгоритм Крускала — MST на графе компонент
    //
    //  Сортируем рёбра по весу (Δ) и жадно добавляем, объединяя компоненты.
    //  Нужно k−1 рёбер, чтобы все подциклы слились в один эйлеров цикл.
    //
    //  extra = суммарная доплата за объединение подциклов
    // ──────────────────────────────────────────────────────────────────────
    std::sort(edges.begin(), edges.end());
    DSU dsuComp(k);

    long double extra = 0;
    for (const auto& e : edges) {
        if (dsuComp.unite(e.u, e.v)) {
            extra += e.w;   // это ребро объединяет два подцикла
        }
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Ответ: base + extra
    //
    //  base  — минимальная сумма поворотов при жадном спаривании
    //          (может давать несколько подциклов)
    //  extra — минимальная доплата за объединение подциклов в один
    //          эйлеров цикл через переключение спариваний в вершинах
    //          степени 4 (найдено через MST на графе компонент)
    // ──────────────────────────────────────────────────────────────────────
    std::cout << std::fixed << std::setprecision(6) << (double)(base + extra) << '\n';
    return 0;
}
```

## C.♟ MindNet — Протокол Исключения

```
#include <iostream>
#include <queue>
#include <vector>

using namespace std;


vector<vector<int>> adj;
vector<int> match_L;
vector<int> match_R;
vector<bool> used;

// ==========================================================================
//  DFS алгоритма Куна
//
//  Пытается найти увеличивающую цепь (augmenting path), начиная с
//  левой вершины v. Увеличивающая цепь — это чередующийся путь,
//  начинающийся и заканчивающийся в свободных вершинах.
//
//  Если цепь найдена, паросочетание "перебрасывается" вдоль неё
//  (matched ↔ unmatched), увеличивая размер паросочетания на 1.
//
//  Возвращает true, если увеличивающая цепь найдена.
// ==========================================================================
bool dfs(int v) {
    // Перебираем всех правых соседей левой вершины v
    for (int to : adj[v]) {
        if (!used[to]) {
            used[to] = true;

            // Если правая вершина to свободна — нашли конец увеличивающей цепи.
            // Если to занята — пробуем рекурсивно "перебросить" её партнёра.
            if (match_R[to] == -1 || dfs(match_R[to])) {
                // Обновляем паросочетание: связываем v ↔ to
                match_L[v] = to;
                match_R[to] = v;
                return true;
            }
        }
    }
    return false;   // увеличивающую цепь из v найти не удалось
}

// ==========================================================================
//  BFS из множества стартовых вершин в ориентированном графе
//
//  Помечает все вершины, достижимые из starts, в массиве reachable.
//  Используется для нахождения множеств A и B (см. теорию выше).
// ==========================================================================
void BFS(const vector<vector<int>>& graph, const vector<int>& starts,
         vector<bool>& reachable) {
    queue<int> q;

    // Инициализация: все стартовые вершины — достижимы
    for (int s : starts) {
        reachable[s] = true;
        q.push(s);
    }

    // Стандартный BFS по ориентированному графу
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int u : graph[v]) {
            if (!reachable[u]) {
                reachable[u] = true;
                q.push(u);
            }
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    // ------------------------------------------------------------------
    //  Чтение входных данных
    //  n1 — вершин в левой доле, n2 — в правой, m — рёбер
    // ------------------------------------------------------------------
    int n1, n2, m;
    cin >> n1 >> n2 >> m;

    adj.resize(n1);
    match_L.assign(n1, -1);
    match_R.assign(n2, -1);
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        u--;  // переходим к 0-индексации
        v--;
        adj[u].push_back(v);  // ребро: левая u — правая v
    }

    // ==================================================================
    //  ШАГ 1: Алгоритм Куна — находим максимальное паросочетание
    //
    //  Для каждой левой вершины запускаем DFS, пытаясь найти
    //  увеличивающую цепь. Если нашли — паросочетание растёт на 1.
    //
    //  Сложность: O(n₁) запусков DFS, каждый O(n₂ + m) → O(n₁·(n₂+m))
    // ==================================================================
    for (int v = 0; v < n1; v++) {
        used.assign(n2, false);  // сброс посещений перед каждым запуском
        dfs(v);
    }

    // ==================================================================
    //  ШАГ 2: Построение ориентированного графа D и обратного D_rev
    //
    //  Нумерация вершин в объединённом графе:
    //    [0 .. n1-1]        — левые вершины
    //    [n1 .. n1+n2-1]    — правые вершины
    //
    //  Правила ориентации рёбер:
    //    • Matched-ребро  (l, r) ∈ M:   D содержит r → l
    //      (моделирует "возврат" по паросочетанию в чередующемся пути)
    //    • Unmatched-ребро (l, r) ∉ M:  D содержит l → r
    //      (моделирует "продвижение" по свободному ребру)
    //
    //  D_rev — граф с обращёнными рёбрами, нужен для анализа правой доли
    // ==================================================================
    int total = n1 + n2;   // общее число вершин в объединённом графе
    vector<vector<int>> D(total), D_rev(total);

    for (int l = 0; l < n1; l++) {
        for (int r : adj[l]) {
            if (match_L[l] == r) {
                // ── Matched-ребро (l, r) ──
                // В D: правая → левая (r → l), моделируя "шаг назад" по M
                D[n1 + r].push_back(l);
                D_rev[l].push_back(n1 + r);
            } else {
                // ── Unmatched-ребро (l, r) ──
                // В D: левая → правая (l → r), моделируя "шаг вперёд" не по M
                D[l].push_back(n1 + r);
                D_rev[n1 + r].push_back(l);
            }
        }
    }

    // ==================================================================
    //  ШАГ 3: BFS из свободных ЛЕВЫХ вершин в D → множество A
    //
    //  A содержит все вершины, до которых можно добраться из свободных
    //  левых вершин по чередующимся путям.
    //
    //  Если покрытая левая вершина l попала в A — значит существует
    //  чередующийся путь от свободной вершины до l, и можно "перебросить"
    //  паросочетание, освободив l. Следовательно, l НЕ обязательна.
    // ==================================================================
    vector<int> free_L, free_R;

    for (int l = 0; l < n1; l++)
        if (match_L[l] == -1)
            free_L.push_back(l);          // свободная левая вершина

    for (int r = 0; r < n2; r++)
        if (match_R[r] == -1)
            free_R.push_back(n1 + r);     // свободная правая вершина

    vector<bool> reachA(total, false);     // достижимость из свободных левых
    vector<bool> reachB(total, false);     // достижимость из свободных правых

    BFS(D, free_L, reachA);

    // ==================================================================
    //  ШАГ 4: BFS из свободных ПРАВЫХ вершин в D_rev → множество B
    //
    //  D_rev обращает все рёбра D, и BFS из свободных правых вершин
    //  в D_rev эквивалентен поиску чередующихся путей, начинающихся
    //  в свободных правых вершинах и идущих "в обратном направлении".
    //
    //  Если покрытая правая вершина r попала в B — она не обязательна.
    // ==================================================================
    BFS(D_rev, free_R, reachB);

    for (int l = 0; l < n1; l++) {
        if (match_L[l] != -1 && !reachA[l]) {
            // l в каждом макс. паросочетании → первый игрок (Коля) побеждает
            cout << 'N';
        } else {
            // l можно освободить (или она и так свободна) → Петя побеждает
            cout << 'P';
        }
    }
    cout << '\n';

    for (int r = 0; r < n2; r++) {
        if (match_R[r] != -1 && !reachB[n1 + r]) {
            // r в каждом макс. паросочетании → Коля побеждает
            cout << 'N';
        } else {
            // r не обязательна → Петя побеждает
            cout << 'P';
        }
    }
    cout << '\n';

    return 0;
}
```

## D. ⚔️ Битва за Влияние

```txt
/*
 * ============================================================================
 *  Задача: Baseball Elimination (Битва за Влияние)
 * ============================================================================
 *
 *  АСИМПТОТИКА:
 *  ─────────────────────────────────────────────────────────────────────────
 *  Построение сети:
 *    • Узлов V = O(n²) — до C(n-1,2) игровых узлов + (n-1) командных + 2
 *    • Рёбер E = O(n²) — по 3 ребра на каждую игровую пару + (n-1) к стоку
 *
 *  Алгоритм Эдмондса-Карпа (BFS-поиск увеличивающих путей):
 *    • Гарантия: не более O(V·E) увеличивающих путей
 *    • Каждый BFS: O(E)
 *    • Итого: O(V · E²) = O(n² · (n²)²) = O(n⁶)
 *
*/
#include <climits>
#include <iostream>
#include <queue>
#include <vector>

using namespace std;

struct Edge {
    int to, cap, flow;
};

vector<Edge> edges;
vector<vector<int>> graph;
int total_nodes;

void AddEdge(int from, int to, int cap) {
    graph[from].push_back(edges.size());         // индекс прямого ребра
    edges.push_back({to, cap, 0});               // прямое ребро

    graph[to].push_back(edges.size());           // индекс обратного ребра
    edges.push_back({from, 0, 0});               // обратное ребро (ёмкость 0)
}

bool BFS(int source, int sink, vector<int>& parent_edge) {
    vector<bool> visited(total_nodes, false);
    queue<int> q;
    q.push(source);
    visited[source] = true;
    parent_edge[source] = -1;
    
    while (!q.empty()) {
        int v = q.front();
        q.pop();
        for (int id : graph[v]) {
            int u = edges[id].to;
            if (!visited[u] && edges[id].cap - edges[id].flow > 0) {
                visited[u] = true;
                parent_edge[u] = id;
                if (u == sink) return true;
                q.push(u);
            }
        }
    }
    return false;
}

// ==========================================================================
//  Алгоритм Эдмондса-Карпа (Ford-Fulkerson с BFS).
//
//  Пока существует увеличивающий путь source → sink:
//    1. Находим минимальную остаточную ёмкость вдоль пути (bottleneck)
//    2. Прибавляем bottleneck к потоку прямых рёбер на пути
//    3. Вычитаем bottleneck из потока обратных рёбер (для возможности отмены)
//
// ==========================================================================

int MaxFlow(int source, int sink) {
    int flow = 0;
    vector<int> parent_edge(total_nodes);

    // Повторяем, пока BFS находит увеличивающий путь
    while (BFS(source, sink, parent_edge)) {

        // Шаг 1: Находим bottleneck — минимальную остаточную ёмкость на пути
        int path_flow = INT_MAX;
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            path_flow = min(path_flow, edges[id].cap - edges[id].flow);
            v = edges[id ^ 1].to;   // переходим к предыдущей вершине на пути
        }

        // Шаг 2: Обновляем потоки вдоль найденного пути
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            edges[id].flow += path_flow;       // увеличиваем поток прямого ребра
            edges[id ^ 1].flow -= path_flow;   // уменьшаем поток обратного ребра
            v = edges[id ^ 1].to;              // переходим к предыдущей вершине
        }

        flow += path_flow;   // добавляем поток пути к общему потоку
    }
    return flow;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n; cin >> n;

    vector<int> wins(n), remaining(n);

    for (int i = 0; i < n; i++) cin >> wins[i];
    for (int i = 0; i < n; i++) cin >> remaining[i];
    vector<vector<int>> matrix(n, vector<int>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> matrix[i][j];

    int max_wins = wins[0] + remaining[0];

    for (int i = 1; i < n; i++) {
        if (wins[i] > max_wins) {
            cout << "NO" << "\n";
            return 0;
        }
    }

    vector<pair<int, int>> game_pairs;
    for (int i = 1; i < n; i++)
        for (int j = i + 1; j < n; j++)
            if (matrix[i][j] > 0)
                game_pairs.push_back({i, j});

    int num_games = game_pairs.size();   // количество игровых узлов
    
    int source = 0;
    int sink = 1 + num_games + (n - 1);
    total_nodes = sink + 1;

    graph.assign(total_nodes, {});
    edges.clear();

    int total_flow_needed = 0;

    for (int g = 0; g < num_games; g++) {
        int i = game_pairs[g].first;
        int j = game_pairs[g].second;

        int game_node = 1 + g;
        int team_i = 1 + num_games + (i - 1);
        int team_j = 1 + num_games + (j - 1);

        AddEdge(source, game_node, matrix[i][j]);
        AddEdge(game_node, team_i, matrix[i][j]);
        AddEdge(game_node, team_j, matrix[i][j]);

        total_flow_needed += matrix[i][j];
    }


    for (int i = 1; i < n; i++) {
        int team_node = 1 + num_games + (i - 1);
        AddEdge(team_node, sink, max_wins - wins[i]);
    }

    int flow = MaxFlow(source, sink);
    cout << (flow == total_flow_needed ? "YES" : "NO") << "\n";

    return 0;
}
```

## E. 📜 GrayZone — Искусство на выживание

```
/*
 * ============================================================================
 *  GrayZone — Минимальная стоимость перекраски + серых линий
 *             Решение: Min-Cut через алгоритм Эдмондса-Карпа
 * ============================================================================
 *
 *  АСИМПТОТИКА:
 *  ─────────────────────────────────────────────────────────────────────────
 *  P = N × M — количество пикселей (до 4900)
 *  V = P + 2 — вершин в сети (пиксели + source + sink)
 *  E = O(P)  — рёбер (≈5P: до 4 соседей + 1 к source/sink на пиксель)
 *
 *  Этап                              │ Сложность
 *  ──────────────────────────────────-┼──────────────────────────
 *  Построение сети                   │ O(P)
 *  Эдмондс-Карп (теор. оценка)      │ O(V · E²) = O(P³)
 *  ──────────────────────────────────-┼──────────────────────────
 *  При N, M ≤ 70: P ≤ 4900          │ На практике быстрее ≈ 10⁶–10⁷
 *  ─────────────────────────────────────────────────────────────────────────
 *
 *  МОДЕЛЬ MIN-CUT:
 *  ─────────────────────────────────────────────────────────────────────────
 *
 *  Стоимости:
 *    • Перекраска B → W: w       • Перекраска W → B: b
 *    • Оставить как есть: 0      • Серая линия: g за каждую пару соседей
 *                                                   с разным финальным цветом
 *
 *  Разрез разбивает пиксели на две группы:
 *    S-сторона (белые) и T-сторона (чёрные).
 *
 *  Ребро попадает в разрез (и платится его ёмкость), если:
 *    • S→p разрезано: p на T-стороне (стал чёрным, хотя был белым) → cap = b
 *    • p→T разрезано: p на S-стороне (стал белым, хотя был чёрным) → cap = w
 *    • p→q разрезано: p белый, q чёрный (серая линия)               → cap = g
 *
 *  ┌────────┐   b (если W)    ┌───────┐   w (если B)    ┌──────┐
 *  │ SOURCE │ ──────────────► │ pixel │ ──────────────► │ SINK │
 *  │  (W)   │                 │       │                 │  (B) │
 *  └────────┘                 └───┬───┘                 └──────┘
 *                                 │
 *                          g ◄────┴────► g
 *                              neighbor
 *  ─────────────────────────────────────────────────────────────────────────
 */

#include <climits>
#include <iostream>
#include <queue>
#include <vector>

using namespace std;

// ==========================================================================
//  Структура ребра потоковой сети
//  to   — вершина-назначение
//  cap  — пропускная способность (максимальный допустимый поток)
//  flow — текущий поток через ребро
// ==========================================================================
struct Edge {
    int to, cap, flow;
};

// edges  — массив ВСЕХ рёбер (прямых и обратных), хранятся парами:
//          edges[2k] — прямое ребро, edges[2k+1] — обратное.
//          Благодаря этому edges[id ^ 1] — парное ребро для edges[id].
// graph  — graph[v] содержит ИНДЕКСЫ рёбер в массиве edges, исходящих из v
vector<Edge> edges;
vector<vector<int>> graph;
int total_nodes;

// ==========================================================================
//  Добавление ребра from→to с пропускной способностью cap.
//  Одновременно создаётся обратное ребро to→from с ёмкостью 0.
//
//  Обратное ребро необходимо для алгоритма: оно позволяет "отменять"
//  ранее пущенный поток, перенаправляя его по другому пути.
//
//  Пример: после AddEdge(A, B, 10):
//    edges[2k]   = {to=B, cap=10, flow=0}  — прямое
//    edges[2k+1] = {to=A, cap=0,  flow=0}  — обратное
// ==========================================================================
void AddEdge(int from, int to, int cap) {
    graph[from].push_back(edges.size());   // индекс прямого ребра
    edges.push_back({to, cap, 0});         // прямое ребро

    graph[to].push_back(edges.size());     // индекс обратного ребра
    edges.push_back({from, 0, 0});         // обратное ребро (ёмкость 0)
}

// ==========================================================================
//  BFS — поиск в ширину кратчайшего увеличивающего пути source → sink
//  в ОСТАТОЧНОЙ сети (только по рёбрам с cap - flow > 0).
//
//  Ключевая гарантия Эдмондса-Карпа: использование BFS (а не DFS) даёт
//  полиномиальную оценку O(VE) на число увеличивающих путей, потому что
//  длина кратчайшего пути может только расти (не уменьшаться) между
//  итерациями.
//
//  parent_edge[v] = индекс ребра, по которому BFS пришёл в вершину v.
//  Возвращает true, если путь до sink найден.
// ==========================================================================
bool BFS(int source, int sink, vector<int>& parent_edge) {
    vector<bool> visited(total_nodes, false);
    queue<int> q;
    q.push(source);
    visited[source] = true;
    parent_edge[source] = -1;

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        // Перебираем все рёбра из v в остаточной сети
        for (int id : graph[v]) {
            int u = edges[id].to;
            int residual = edges[id].cap - edges[id].flow;

            // Переходим только по рёбрам с положительной остаточной ёмкостью
            if (!visited[u] && residual > 0) {
                visited[u] = true;
                parent_edge[u] = id;

                // Ранний выход: путь до стока найден, дальнейший BFS не нужен
                if (u == sink) return true;
                q.push(u);
            }
        }
    }
    return false;   // стока не достичь → поток максимален
}

// ==========================================================================
//  Алгоритм Эдмондса-Карпа (Ford-Fulkerson с BFS)
//
//  Цикл:
//    1. BFS находит кратчайший увеличивающий путь source → sink
//    2. Определяем bottleneck — мин. остаточную ёмкость на пути
//    3. Проталкиваем bottleneck единиц потока:
//       • прямые рёбра: flow += bottleneck
//       • обратные рёбра: flow -= bottleneck (открывают возможность отмены)
//    4. Повторяем, пока BFS находит путь
//
//  Теорема: макс. поток = мин. разрез (Ford-Fulkerson)
// ==========================================================================
int MaxFlow(int source, int sink) {
    int flow = 0;
    vector<int> parent_edge(total_nodes);

    while (BFS(source, sink, parent_edge)) {
        // ── Шаг 1: Находим bottleneck (узкое место) ──
        // Идём от sink назад к source по рёбрам, запомненным BFS
        int path_flow = INT_MAX;
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            path_flow = min(path_flow, edges[id].cap - edges[id].flow);
            v = edges[id ^ 1].to;   // id^1 — обратное ребро, его .to — начало прямого
        }

        // ── Шаг 2: Проталкиваем поток по пути ──
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            edges[id].flow += path_flow;       // прямое ребро: +поток
            edges[id ^ 1].flow -= path_flow;   // обратное ребро: -поток
            v = edges[id ^ 1].to;
        }

        flow += path_flow;
    }
    return flow;   // = минимальный разрез
}

// ==========================================================================
//  main()
// ==========================================================================
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    // ──────────────────────────────────────────────────────────────────
    //  Чтение входных данных
    //  N × M — размеры полотна
    //  w     — стоимость перекраски чёрного пикселя в белый  (B → W)
    //  b     — стоимость перекраски белого пикселя в чёрный  (W → B)
    //  g     — стоимость одной серой линии (между соседями разного цвета)
    // ──────────────────────────────────────────────────────────────────
    int N, M, w, b, g;
    cin >> N >> M >> w >> b >> g;

    vector<string> grid(N);
    for (int i = 0; i < N; i++) cin >> grid[i];

    // ──────────────────────────────────────────────────────────────────
    //  Нумерация вершин потоковой сети:
    //
    //    0              — Source (олицетворяет "белый цвет")
    //    i * M + j + 1  — пиксель на позиции (i, j), нумерация с 1
    //    N * M + 1      — Sink   (олицетворяет "чёрный цвет")
    //
    //  Всего вершин: N * M + 2
    // ──────────────────────────────────────────────────────────────────
    int source = 0;
    int sink = N * M + 1;
    total_nodes = N * M + 2;
    graph.resize(total_nodes);

    // Лямбда: координаты (i, j) → номер вершины в сети
    auto idx = [&](int i, int j) -> int { return i * M + j + 1; };

    // ──────────────────────────────────────────────────────────────────
    //  Построение сети
    //
    //  Для каждого пикселя (i, j):
    //
    //  ► Если пиксель ЧЁРНЫЙ ('B'):
    //      • Остаётся чёрным (T-сторона) — бесплатно, ребро S→p не нужно
    //      • Перекрашивается в белый (S-сторона) — стоит w
    //        → добавляем p → Sink с ёмкостью w
    //        (если p окажется на S-стороне, это ребро попадёт в разрез)
    //
    //  ► Если пиксель БЕЛЫЙ ('W'):
    //      • Остаётся белым (S-сторона) — бесплатно, ребро p→T не нужно
    //      • Перекрашивается в чёрный (T-сторона) — стоит b
    //        → добавляем Source → p с ёмкостью b
    //        (если p окажется на T-стороне, это ребро попадёт в разрез)
    //
    //  ► Между каждой парой соседних пикселей:
    //      • Если в финале у них разный цвет — серая линия стоит g
    //      • Добавляем ДВА направленных ребра p↔q, каждое с ёмкостью g
    //      • Если p на S-стороне, q на T-стороне: ребро p→q в разрезе (g)
    //      • Если q на S-стороне, p на T-стороне: ребро q→p в разрезе (g)
    //      • Оба на одной стороне: ни одно не в разрезе (0) ✓
    // ──────────────────────────────────────────────────────────────────
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            int p = idx(i, j);

            if (grid[i][j] == 'B') {
                // ── Чёрный пиксель ──
                // Перекраска B→W стоит w: ребро p → Sink
                AddEdge(p, sink, w);
            } else {
                // ── Белый пиксель ──
                // Перекраска W→B стоит b: ребро Source → p
                AddEdge(source, p, b);
            }

            // ── Рёбра к соседям (серые линии) ──
            // Обрабатываем только соседа СНИЗУ и СПРАВА, чтобы каждая пара
            // обрабатывалась ровно один раз. Но добавляем ОБА направления!

            // Сосед снизу (i+1, j)
            if (i + 1 < N) {
                int q = idx(i + 1, j);
                AddEdge(p, q, g);   // p → q: стоимость g если p белый, q чёрный
                AddEdge(q, p, g);   // q → p: стоимость g если q белый, p чёрный
            }

            // Сосед справа (i, j+1)
            if (j + 1 < M) {
                int q = idx(i, j + 1);
                AddEdge(p, q, g);   // p → q
                AddEdge(q, p, g);   // q → p
            }
        }
    }

    // ──────────────────────────────────────────────────────────────────
    //  Ответ: минимальный разрез = максимальный поток
    //
    //  По теореме Форда-Фалкерсона:
    //    max flow = min cut
    //
    //  Min cut = сумма ёмкостей рёбер, пересекающих разрез
    //          = (стоимость всех перекрасок) + (стоимость всех серых линий)
    //          = МИНИМАЛЬНАЯ стоимость финальной картины
    // ──────────────────────────────────────────────────────────────────
    cout << MaxFlow(source, sink) << "\n";

    return 0;
}
```

## F. 🧾 LoadBalancer

```
/*
 * ============================================================================
 *  LoadBalancer — Максимальная прибыль с учётом скидок
 *                 Решение: Project Selection → Min-Cut (Эдмондс-Карп)
 * ============================================================================
 *
 *  АСИМПТОТИКА:
 *  ─────────────────────────────────────────────────────────────────────────
 *  K = Σ Kᵢ — суммарное число зависимостей "доставка → модуль"
 *
 *  V = N + M + P + 2     (доставки + модули + скидки + source + sink)
 *  E = N + K + M + 3P    (рёбра сети, без обратных)
 *
 *  При N, M ≤ 100, P ≤ 50:   V ≤ 252,  E ≤ ~10 350
 *
 *  Этап                              │ Сложность
 *  ──────────────────────────────────-┼──────────────────────────
 *  Построение сети                   │ O(N + K + M + P)
 *  Эдмондс-Карп                     │ O(V · E²) ≈ O(2.7 · 10¹⁰) теор.
 *                                    │ На практике << из-за структуры графа
 *  ─────────────────────────────────────────────────────────────────────────
 *
 *  МОДЕЛЬ PROJECT SELECTION → MIN-CUT:
 *  ─────────────────────────────────────────────────────────────────────────
 *
 *  Имеем три типа "проектов":
 *
 *  1) ДОСТАВКИ (приносят прибыль Vᵢ, требуют модули)
 *     • Это "положительные проекты" — мы хотим их выбрать
 *     • Выбор доставки i принуждает купить все её модули
 *
 *  2) МОДУЛИ (стоят Cⱼ — это расход)
 *     • Это "отрицательные проекты" — мы вынуждены их покупать
 *
 *  3) СКИДКИ (экономят sₖ = Cₐ + C_b - d, если оба модуля куплены)
 *     • Это "положительные проекты" — бонус за покупку пары
 *     • Применение скидки k требует покупки обоих модулей a и b
 *
 *  Сведение к Min-Cut (S-сторона = "выбрано", T-сторона = "не выбрано"):
 *
 *  ┌────────┐  Vᵢ     ┌──────────┐  INF    ┌──────────┐  Cⱼ    ┌──────┐
 *  │ SOURCE │────────►│ Доставка │───────►│  Модуль  │───────►│ SINK │
 *  │        │         │    i     │        │    j     │        │      │
 *  └────────┘         └──────────┘        └──────────┘        └──────┘
 *       │                                      ▲
 *       │  sₖ    ┌──────────┐      INF         │
 *       └───────►│  Скидка  │──────────────────┘
 *                │    k     │──────────────────► модуль b
 *                └──────────┘      INF
 *
 *  Что означает разрез каждого ребра:
 *  ─────────────────────────────────────────────────────────────
 *  │ Ребро              │ В разрезе когда           │ Цена   │
 *  │────────────────────-│───────────────────────────│────────│
 *  │ S → Доставка i     │ Доставка НЕ выбрана      │ Vᵢ     │
 *  │ S → Скидка k       │ Скидка НЕ применена       │ sₖ     │
 *  │ Доставка → Модуль  │ Никогда (INF)             │ —      │
 *  │ Скидка → Модуль    │ Никогда (INF)             │ —      │
 *  │ Модуль j → T       │ Модуль куплен             │ Cⱼ     │
 *  ─────────────────────────────────────────────────────────────
 *
 *  Ответ = (Σ Vᵢ + Σ sₖ) − MinCut = total_profit − MaxFlow
 *
 *  Корректность:
 *  • INF-рёбра гарантируют: если доставка выбрана (S-сторона),
 *    то все её модули тоже на S-стороне → модуль→T в разрезе → куплен
 *  • Аналогично для скидок: если скидка применена, оба модуля куплены
 *  • Модуль покупается ОДИН раз, даже если нужен нескольким доставкам
 *  • Можно не выбирать ничего (разрезаем все S→ рёбра), прибыль = 0
 * ============================================================================
 */

#include <climits>
#include <iostream>
#include <queue>
#include <vector>

using namespace std;

// ==========================================================================
//  Структура ребра потоковой сети
//  to   — куда ведёт ребро
//  cap  — пропускная способность
//  flow — текущий поток
// ==========================================================================
struct Edge {
    int to, cap, flow;
};

// edges  — все рёбра (прямые и обратные), хранятся парами:
//          edges[2k] прямое, edges[2k+1] обратное → edges[id^1] = парное
// graph  — graph[v] = список индексов рёбер, исходящих из вершины v
vector<Edge> edges;
vector<vector<int>> graph;
int total_nodes;

// ==========================================================================
//  AddEdge — добавляет пару (прямое + обратное ребро)
//
//  Обратное ребро с cap=0 необходимо для алгоритма: оно позволяет
//  "отменить" ранее пущенный поток и перенаправить его по другому пути.
//  Это ключевой механизм, обеспечивающий оптимальность Ford-Fulkerson.
// ==========================================================================
void AddEdge(int from, int to, int cap) {
    graph[from].push_back(edges.size());     // запоминаем индекс прямого ребра
    edges.push_back({to, cap, 0});           // прямое ребро

    graph[to].push_back(edges.size());       // индекс обратного ребра
    edges.push_back({from, 0, 0});           // обратное ребро (cap = 0)
}

// ==========================================================================
//  BFS — поиск кратчайшего увеличивающего пути source → sink
//
//  Работает в ОСТАТОЧНОЙ сети: проходит только по рёбрам с cap-flow > 0.
//  Использование BFS (а не DFS) — ключевое отличие Эдмондса-Карпа от
//  базового Ford-Fulkerson. Гарантирует, что длина кратчайшего пути
//  не уменьшается между итерациями → полиномиальная сложность.
//
//  parent_edge[v] — индекс ребра, по которому BFS пришёл в v.
//  Используется для восстановления пути и определения bottleneck.
// ==========================================================================
bool BFS(int source, int sink, vector<int>& parent_edge) {
    vector<bool> visited(total_nodes, false);
    queue<int> q;
    q.push(source);
    visited[source] = true;
    parent_edge[source] = -1;

    while (!q.empty()) {
        int v = q.front();
        q.pop();

        for (int id : graph[v]) {
            int u = edges[id].to;
            int residual = edges[id].cap - edges[id].flow;

            // Переходим только по рёбрам с положительной остаточной ёмкостью
            if (!visited[u] && residual > 0) {
                visited[u] = true;
                parent_edge[u] = id;

                // Ранний выход: нашли путь до стока
                if (u == sink) return true;
                q.push(u);
            }
        }
    }
    return false;   // стока не достичь → поток уже максимален
}

// ==========================================================================
//  MaxFlow — алгоритм Эдмондса-Карпа
//
//  Повторяем:
//    1. BFS находит кратчайший увеличивающий путь
//    2. Определяем bottleneck — мин. остаточную ёмкость вдоль пути
//    3. Проталкиваем bottleneck единиц потока:
//       прямые рёбра: flow += bottleneck
//       обратные:     flow -= bottleneck (открывает возможность отмены)
//
//  По теореме Ford-Fulkerson: MaxFlow = MinCut
// ==========================================================================
int MaxFlow(int source, int sink) {
    int flow = 0;
    vector<int> parent_edge(total_nodes);

    while (BFS(source, sink, parent_edge)) {
        // ── Bottleneck: минимальная остаточная ёмкость на пути ──
        int path_flow = INT_MAX;
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            path_flow = min(path_flow, edges[id].cap - edges[id].flow);
            v = edges[id ^ 1].to;    // id^1 = обратное ребро → его .to = начало прямого
        }

        // ── Проталкиваем поток вдоль пути ──
        for (int v = sink; v != source;) {
            int id = parent_edge[v];
            edges[id].flow += path_flow;         // прямое: +поток
            edges[id ^ 1].flow -= path_flow;     // обратное: −поток
            v = edges[id ^ 1].to;
        }

        flow += path_flow;
    }
    return flow;
}

// ==========================================================================
//  main()
// ==========================================================================
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    // ──────────────────────────────────────────────────────────────────────
    //  Чтение: N доставок, M модулей, P скидок
    // ──────────────────────────────────────────────────────────────────────
    int N, M, P;
    cin >> N >> M >> P;

    // ──────────────────────────────────────────────────────────────────────
    //  Нумерация вершин потоковой сети:
    //
    //    0                    — Source (олицетворяет "выбрать / применить")
    //    [1 .. N]             — узлы доставок
    //    [N+1 .. N+M]         — узлы модулей  (модуль j → вершина N+j)
    //    [N+M+1 .. N+M+P]    — узлы скидок
    //    N+M+P+1              — Sink (олицетворяет "не выбирать")
    // ──────────────────────────────────────────────────────────────────────
    int source = 0;
    int sink = N + M + P + 1;
    total_nodes = sink + 1;
    graph.resize(total_nodes);

    const int INF = 1e9;

    // total_profit — сумма всех "доходов" (доставки + скидки)
    // Из неё вычтем min_cut, чтобы получить макс. чистую прибыль
    int total_profit = 0;

    // ──────────────────────────────────────────────────────────────────────
    //  Доставки: каждая — "положительный проект"
    //
    //  Source → Доставка_i : cap = Vᵢ
    //    • Если i на S-стороне (выбрана) — ребро НЕ в разрезе, прибыль Vᵢ
    //    • Если i на T-стороне (отвергнута) — ребро В разрезе, теряем Vᵢ
    //
    //  Доставка_i → Модуль_j : cap = INF (для каждого модуля j из Kᵢ)
    //    • INF гарантирует: если доставка выбрана (S-сторона),
    //      все её модули тоже на S-стороне (иначе разрез = ∞)
    //    • Это моделирует ЗАВИСИМОСТЬ: доставка требует модули
    // ──────────────────────────────────────────────────────────────────────
    for (int i = 1; i <= N; i++) {
        int V, K;
        cin >> V >> K;

        total_profit += V;                 // добавляем потенциальный доход
        AddEdge(source, i, V);             // Source → Доставка: cap = Vᵢ

        for (int t = 0; t < K; t++) {
            int mod;
            cin >> mod;
            AddEdge(i, N + mod, INF);      // Доставка → Модуль: cap = INF
        }
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Модули: каждый — "отрицательный проект" (расход)
    //
    //  Модуль_j → Sink : cap = Cⱼ
    //    • Если модуль на S-стороне (куплен) — ребро В разрезе, платим Cⱼ
    //    • Если модуль на T-стороне (не куплен) — ребро НЕ в разрезе
    //
    //  Смысл: покупка модуля стоит Cⱼ, и эта стоимость учитывается в разрезе
    // ──────────────────────────────────────────────────────────────────────
    vector<int> cost(M + 1);
    for (int j = 1; j <= M; j++) {
        cin >> cost[j];
        AddEdge(N + j, sink, cost[j]);     // Модуль → Sink: cap = Cⱼ
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Скидки: каждая — "положительный проект" (бонус)
    //
    //  Скидка (a, b, d): покупка a и b вместе стоит d вместо Cₐ + C_b
    //  Экономия sₖ = Cₐ + C_b − d
    //
    //  Source → Скидка_k : cap = sₖ
    //    • Если скидка на S-стороне (применена) — бонус sₖ не теряется
    //    • Если на T-стороне (не применена) — ребро в разрезе, теряем sₖ
    //
    //  Скидка_k → Модуль_a : cap = INF
    //  Скидка_k → Модуль_b : cap = INF
    //    • Применение скидки ТРЕБУЕТ покупки обоих модулей
    //    • INF-ребро не даст скидке быть на S-стороне, если модуль на T
    //
    //  Если sₖ ≤ 0 — скидка невыгодна, просто игнорируем её
    // ──────────────────────────────────────────────────────────────────────
    for (int k = 0; k < P; k++) {
        int a, b, d;
        cin >> a >> b >> d;

        int savings = cost[a] + cost[b] - d;   // экономия от скидки

        if (savings > 0) {
            int disc_node = N + M + 1 + k;     // номер вершины скидки

            total_profit += savings;                    // потенциальный бонус
            AddEdge(source, disc_node, savings);        // Source → Скидка: cap = sₖ
            AddEdge(disc_node, N + a, INF);             // Скидка → Модуль a: INF
            AddEdge(disc_node, N + b, INF);             // Скидка → Модуль b: INF
        }
    }

    // ──────────────────────────────────────────────────────────────────────
    //  Ответ: MaxProfit = total_profit − MinCut = total_profit − MaxFlow
    //
    //  Почему это корректно:
    //  • total_profit = Σ Vᵢ + Σ sₖ — максимум, если выбрать ВСЁ бесплатно
    //  • MinCut вычитает:
    //      - Vᵢ за каждую отвергнутую доставку    (S→доставка в разрезе)
    //      - sₖ за каждую неприменённую скидку     (S→скидка в разрезе)
    //      - Cⱼ за каждый купленный модуль         (модуль→T в разрезе)
    //  • MinCut МИНИМИЗИРУЕТ эти потери → оставшаяся прибыль МАКСИМАЛЬНА
    //
    //  Гарантия неотрицательности:
    //  Разрез "отвергнуть всё" (все S→ рёбра) = total_profit
    //  ⟹ MinCut ≤ total_profit ⟹ ответ ≥ 0
    // ──────────────────────────────────────────────────────────────────────
    int min_cut = MaxFlow(source, sink);

    cout << total_profit - min_cut << "\n";

    return 0;
}
```
